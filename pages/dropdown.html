Testing Tips

// Selenium (Python)
from selenium.webdriver.support.ui import Select
dropdown = Select(driver.find_element(By.ID, "dropdown"))

# By visible text
dropdown.select_by_visible_text("Option 1")

# By value
dropdown.select_by_value("1")

# By index
dropdown.select_by_index(0)
