Testing Tips

// Handle alert
driver.find_element(By.ID, "js-alert").click()
alert = driver.switch_to.alert
alert.accept()

// Handle confirm (accept or dismiss)
alert = driver.switch_to.alert
alert.accept()  # OK
# or alert.dismiss()  # Cancel

// Handle prompt
alert = driver.switch_to.alert
alert.send_keys("My Text")
alert.accept()
