Testing Tips

// Set slider value directly
slider = driver.find_element(By.ID, "slider")
driver.execute_script("arguments[0].value = 3", slider)
# Trigger change event
driver.execute_script("arguments[0].dispatchEvent(new Event('input'))", slider)

// Using keyboard
slider.click()
slider.send_keys(Keys.ARROW_RIGHT)  # Increment
slider.send_keys(Keys.ARROW_LEFT)   # Decrement

// Using ActionChains
actions = ActionChains(driver)
actions.click_and_hold(slider).move_by_offset(50, 0).release().perform()
