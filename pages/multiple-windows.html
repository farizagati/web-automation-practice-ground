# Get all window handles
handles = driver.window_handles

# Switch to new window
driver.switch_to.window(handles[1])

# Do something in new window
# ...

# Switch back to original
driver.switch_to.window(handles[0])

# Close new window
driver.switch_to.window(handles[1])
driver.close()
driver.switch_to.window(handles[0])
