Testing Tips

// Scroll to bottom of page
driver.execute_script("window.scrollTo(0, document.body.scrollHeight)")

// Wait for new content to load
initial_count = len(driver.find_elements(By.CLASS_NAME, "jscroll-added"))

# Scroll
driver.execute_script("window.scrollTo(0, document.body.scrollHeight)")

# Wait for count to increase
WebDriverWait(driver, 10).until(
    lambda d: len(d.find_elements(By.CLASS_NAME, "jscroll-added")) > initial_count
)

// Playwright - scroll and wait
await page.evaluate('window.scrollTo(0, document.body.scrollHeight)')
await page.waitForSelector('.jscroll-added:nth-child(5)')
